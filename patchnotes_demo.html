import patchnotes

# parse from string or file
cl = patchnotes.parse_file("CHANGELOG.md")
cl = patchnotes.parse(raw_text)

# access releases
cl.latest()       # → Release(v2.1.0, 2024-11-15, 6 entries)
cl.unreleased()   # → Release(vUnreleased, unreleased, 2 entries)

# iterate all
for release in cl.releases:
    print(release.version, release.release_date)

# get a specific version
r = cl.get_version("2.0.0")

# entries grouped by type
r.by_type
# → {'Breaking': [...], 'Added': [...], 'Security': [...]}

# just breaking / removed entries
r.breaking_changes

# all breaking changes across every version
cl.all_breaking_changes()
# → [('2.0.0', Entry(Breaking: 'Renamed connect()...')), ...]

# releases since a version
cl.since_version("1.4.0")
# → [Release(vUnreleased), Release(v2.1.0), Release(v2.0.0), Release(v1.4.2)]

# what changed between two versions?
releases = cl.diff("1.4.0", "2.1.0")

for r in releases:
    print(f"v{r.version} — {len(r.entries)} changes")
    for entry in r.breaking_changes:
        print(f"  ⚠ BREAKING: {entry.text}")

# v2.1.0 — 6 changes
# v2.0.0 — 8 changes
#   ⚠ BREAKING: Renamed `connect()` to `open_connection()`
#   ⚠ BREAKING: Removed deprecated `legacy_auth` parameter
# v1.4.2 — 4 changes

// Weekend 2 goal: same API, Rust core via PyO3
// patchnotes_core/src/lib.rs

use pyo3::prelude::*;

#[pyclass]
struct Release {
    version: String,
    entries: Vec<Entry>,
}

#[pymethods]
impl Release {
    fn breaking_changes(&self) -> Vec<&Entry> {
        self.entries.iter()
            .filter(|e| e.change_type == ChangeType::Breaking)
            .collect()
    }
}
